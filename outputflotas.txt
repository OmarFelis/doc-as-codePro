    private static final String DEFAULT_TIMEZONE = "UTC";
    private static final Locale DEFAULT_LOCALE = Locale.US;

    // Formato thread-safe
    private static final ThreadLocal<SimpleDateFormat> formatter =
        ThreadLocal.withInitial(() -> {
            SimpleDateFormat sdf = new SimpleDateFormat(DEFAULT_PATTERN, DEFAULT_LOCALE);
            sdf.setTimeZone(TimeZone.getTimeZone(DEFAULT_TIMEZONE));
            return sdf;
        });

     */

            PageContext pageContext,
            MediaTypeEnum media) throws DecoratorException {

        try {
            // Validar entrada
            if (columnValue == null) {
                return "";
            }

            // Validar tipo de dato
            validateDateType(columnValue);

            // Formatear fecha de forma segura
            return formatDateSafely((Date) columnValue);

        } catch (Exception e) {
            handleFormatError(e, columnValue);
            return handleFailedFormat(columnValue);
        }
    }

    /**
     * Validación de tipo de fecha
     */
    private void validateDateType(Object value) throws DecoratorException {
        if (!(value instanceof Date)) {
            throw new DecoratorException(
                "Value must be an instance of java.util.Date"
            );
        }
    }

    /**
     * Formateo seguro de fecha
     */
    private String formatDateSafely(Date date) {
        try {
            // Validar fecha
            validateDateValue(date);

            // Obtener formateador thread-safe
            SimpleDateFormat sdf = formatter.get();

            // Formatear y sanitizar
            String formattedDate = sdf.format(date);
            return sanitizeOutput(formattedDate);

        } finally {
            // Limpiar ThreadLocal si es necesario
            cleanupFormatter();
        }
    }

    /**
     * Validación de valor de fecha
     */
    private void validateDateValue(Date date) throws DecoratorException {
        if (date == null) {
            throw new DecoratorException("Date value cannot be null");
        }

        // Validar rango de fecha
        long now = System.currentTimeMillis();
        long dateTime = date.getTime();

        if (dateTime > now + (365 * 24 * 60 * 60 * 1000L)) { // 1 año en el futuro
            throw new DecoratorException("Date too far in the future");
        }

        if (dateTime < 0) { // Antes de 1970
            throw new DecoratorException("Date too far in the past");
        }
    }

            // Formatear fecha de forma segura
            return formatDateSafely((Date) columnValue);

        } catch (Exception e) {
            handleFormatError(e, columnValue);
            return handleFailedFormat(columnValue);
        } finally {

        long dateTime = date.getTime();

    private BigDecimal parseNumber(String value) {

        try {

            try (OutputStream output = new BufferedOutputStream(

                copyStream(input, output);

            if (!verifyFileIntegrity(tempFile, input)) {

            if (total > MAX_FILE_SIZE) {
                throw new IOException("File size exceeds maximum");
            }
            output.write(buffer, 0, length);
        }

     */
    @Override
    public void init(

            super.init(config);

            // Validar y cargar configuración
            ConfigurationParameters params = validateAndLoadConfiguration(config);

            // Configurar logging
            configureLogging(params);

            LOGGER.info("Logging configuration initialized successfully");

        } catch (Exception e) {

            throw new ServletException("Failed to initialize logging", e);
        }
    }

    /**
     * Destrucción segura del servlet
     */
    @Override
    public void destroy() {
        try {
            LOGGER.info("Initiating LoadProperties shutdown");
            shutdownLogging();
            super.destroy();
            LOGGER.info("LoadProperties shutdown completed");

        } catch (Exception e) {
            handleShutdownError("Error during shutdown", e);
        }
    }

    /**
     * Clase para parámetros de configuración
     */
    private static class ConfigurationParameters {
        private final Path configPath;
        private final boolean watchEnabled;
        private final long watchInterval;

        public ConfigurationParameters(
                Path configPath,
                boolean watchEnabled,
                long watchInterval) {
            this.configPath = configPath;
            this.watchEnabled = watchEnabled;
            this.watchInterval = watchInterval;
        }
    }

    /**
     * Validación y carga de configuración
     */
    private ConfigurationParameters validateAndLoadConfiguration(
            ServletConfig config) throws ServletException {

        // Obtener y validar path base
        String prefix = validateBasePath(
            config.getServletContext().getRealPath("/"));

        // Obtener y validar archivo de configuración
        String configFile = validateConfigFile(
            config.getInitParameter("log4j-conf-file"));

        // Construir y validar path completo
        Path configPath = validateFullPath(prefix, configFile);

        // Validar parámetros de watch
        boolean watchEnabled = validateWatchEnabled(
            config.getInitParameter("watch"));

        long watchInterval = validateWatchInterval(
            config.getInitParameter("time-watch"));

        return new ConfigurationParameters(
            configPath, watchEnabled, watchInterval);
    }

    /**

    private String validateBasePath(String path) throws ServletException {
        if (path == null || path.trim().isEmpty()) {
            throw new ServletException("Base path cannot be empty");
        }

        path = path.replace('\\', '/');
        validatePath(path);

        return path;
    }

    private String validateConfigFile(String filename) throws ServletException {
        if (filename == null || filename.trim().isEmpty()) {
            throw new ServletException("Config filename cannot be empty");
        }

        if (!filename.matches(SAFE_PATH_PATTERN)) {
            throw new ServletException("Invalid config filename format");
        }

        return filename;
    }

                throw new ServletException(
                    "Cannot access config file: " + path);

            if (file.length() > MAX_CONFIG_SIZE) {
                throw new ServletException("Config file too large");
            }

            validatePath(path.toString());
            return path;

        } catch (SecurityException e) {
            throw new ServletException("Security violation accessing config", e);
        }

        // Verificar rutas prohibidas

            }
        }

            if (value < MIN_WATCH_INTERVAL) {
                return MIN_WATCH_INTERVAL;
            }
            if (value > MAX_WATCH_INTERVAL) {
                return MAX_WATCH_INTERVAL;
            }
            return value;

        } catch (NumberFormatException e) {
            return MIN_WATCH_INTERVAL;
        }
    }

        }

        }
    }

    private void handleInitError(String message, Exception e) {
        LOGGER.log(Level.SEVERE, message, e);
    }

                    this.vector.addAll(validatedVector);

            Object value = session.getAttribute(key);
            if (value != null && type.isInstance(value)) {
                return type.cast(value);
            }

            return null;

        } catch (Exception e) {
            handleUtilError("Error getting session object", e);
            return null;
        }
    }

     * Validaciones
     */
    private static void validateArrayInput(String[] array) {
        if (array == null) {
            throw new IllegalArgumentException("Array cannot be null");
        }

        if (array.length > MAX_ARRAY_SIZE) {
            throw new IllegalArgumentException("Array too large");
        }
    }

    private static void validateSeparator(String separator) {
        if (separator == null || separator.trim().isEmpty()) {
            throw new IllegalArgumentException("Separator cannot be empty");
        }

        if (!separator.matches(SAFE_CHARS_PATTERN)) {
            throw new IllegalArgumentException("Invalid separator format");
        }
    }

    private static void validateAmount(BigDecimal amount) {
        if (amount == null) {
            throw new IllegalArgumentException("Amount cannot be null");
        }

        if (amount.compareTo(BigDecimal.ZERO) < 0) {
            throw new IllegalArgumentException("Amount cannot be negative");
        }
    }

    private static void validateRequest(HttpServletRequest request) {
        if (request == null) {
            throw new IllegalArgumentException("Request cannot be null");
        }
    }

    /**
     * Sanitización
     */
    private static String sanitizeInput(String input) {
        if (input == null) {
            return "";
        }
        return Encode.forHtml(input);
    }

    private static String sanitizeNumericInput(String input) {
        if (input == null) {
            return "0";
        }
        return input.replaceAll("[^0-9.-]", "");
    }

    /**
     * Manejo de errores
     */
    private static void handleUtilError(String message, Exception e) {
        LOGGER.log(Level.SEVERE, message, e);
    }
